using System.Diagnostics.Tracing;
using System.Diagnostics;
using System.Text.Json;
using MethodCache.Benchmarks.Microbenchmarks;

var bench = new HttpColdMissBenchmarks { Listening = false, AfterInvalidation = false };
bench.Setup().GetAwaiter().GetResult();
try
{
    for (var batch=0;batch<8;batch++)
    {
        bench.ResetBatch();
        for (var i=0;i<HttpColdMissBenchmarks.InvocationsPerIteration;i++) _=bench.Miss().GetAwaiter().GetResult();
    }
    using var listener = new AllocationListener();
    long total=0;
    var clock=Stopwatch.StartNew();
    const int batches=64;
    for (var batch=0;batch<batches;batch++)
    {
        bench.ResetBatch();
        var before=GC.GetAllocatedBytesForCurrentThread();
        for (var i=0;i<HttpColdMissBenchmarks.InvocationsPerIteration;i++) _=bench.Miss().GetAwaiter().GetResult();
        total+=GC.GetAllocatedBytesForCurrentThread()-before;
    }
    clock.Stop();
    Console.WriteLine($"Diagnostic cold HTTP miss allocation: {total/(double)(batches*HttpColdMissBenchmarks.InvocationsPerIteration)} B/op; elapsed including resets: {clock.Elapsed}");
    // Only allow diagnostic event delivery; this is not a timing or behavior test.
    Thread.Sleep(500);
    Console.WriteLine(JsonSerializer.Serialize(listener.Snapshot()));
}
finally { bench.Cleanup(); }

sealed class AllocationListener : EventListener
{
    private readonly Dictionary<string,long> _samples = new(StringComparer.Ordinal);
    protected override void OnEventSourceCreated(EventSource source)
    {
        if(source.Name=="Microsoft-Windows-DotNETRuntime") EnableEvents(source,EventLevel.Verbose,(EventKeywords)1);
    }
    protected override void OnEventWritten(EventWrittenEventArgs data)
    {
        if(data.EventName?.StartsWith("GCAllocationTick",StringComparison.Ordinal)!=true || data.PayloadNames is null || data.Payload is null) return;
        var type=data.PayloadNames.IndexOf("TypeName");
        if(type<0 || data.Payload[type] is not string name)return;
        lock(_samples) _samples[name]=_samples.GetValueOrDefault(name)+1;
    }
    public object Snapshot()
    {
        lock(_samples) return _samples.OrderByDescending(p=>p.Value).Select(p=>new{type=p.Key,samples=p.Value}).ToArray();
    }
}
